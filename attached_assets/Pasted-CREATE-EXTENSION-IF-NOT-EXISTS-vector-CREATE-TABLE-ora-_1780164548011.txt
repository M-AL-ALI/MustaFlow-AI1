CREATE EXTENSION IF NOT EXISTS "vector";
CREATE TABLE "ora_transcripts" (
	"id" serial PRIMARY KEY NOT NULL,
	"user_id" text NOT NULL,
	"messages" jsonb DEFAULT '[]'::jsonb NOT NULL,
	"updated_at" timestamp with time zone DEFAULT now() NOT NULL,
	CONSTRAINT "ora_transcripts_user_id_key" UNIQUE("user_id")
);
CREATE TABLE "vault_versions" (
	"id" serial PRIMARY KEY NOT NULL,
	"entry_id" integer NOT NULL,
	"version" integer NOT NULL,
	"title" text NOT NULL,
	"summary" text NOT NULL,
	"content" text NOT NULL,
	"tags" text[] DEFAULT '{}' NOT NULL,
	"department" text,
	"edited_by" text NOT NULL,
	"edited_at" timestamp with time zone DEFAULT now() NOT NULL,
	"change_summary" text
);
CREATE TABLE "vault_embeddings" (
	"id" serial PRIMARY KEY NOT NULL,
	"entry_id" integer NOT NULL,
	"user_id" text NOT NULL,
	"chunk_index" integer NOT NULL,
	"chunk_text" text NOT NULL,
	"chunk_hash" text NOT NULL,
	"embedding" vector(1536),
	"embedding_model" text DEFAULT 'text-embedding-3-small' NOT NULL,
	"source_version" integer NOT NULL,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL,
	"updated_at" timestamp with time zone DEFAULT now() NOT NULL,
	CONSTRAINT "vault_embeddings_entry_chunk_unique" UNIQUE("entry_id","chunk_index")
);
CREATE TABLE "vault_entries" (
	"id" serial PRIMARY KEY NOT NULL,
	"user_id" text NOT NULL,
	"title" text NOT NULL,
	"category" text DEFAULT 'OTHER' NOT NULL,
	"subcategory" text,
	"summary" text NOT NULL,
	"content" text NOT NULL,
	"tags" text[] DEFAULT '{}' NOT NULL,
	"department" text,
	"source_type" text DEFAULT 'USER_CREATED' NOT NULL,
	"source_reference" text,
	"status" text DEFAULT 'draft' NOT NULL,
	"version" integer DEFAULT 1 NOT NULL,
	"confidence_score" integer,
	"approved" boolean DEFAULT false NOT NULL,
	"updated_by" text,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL,
	"updated_at" timestamp with time zone DEFAULT now() NOT NULL,
	"archived_at" timestamp with time zone
);

CREATE INDEX "vault_embeddings_user_idx" ON "vault_embeddings" USING btree ("user_id","entry_id");
CREATE INDEX "vault_entries_archived_idx" ON "vault_entries" USING btree ("user_id","archived_at") WHERE (archived_at IS NOT NULL);
CREATE INDEX "vault_entries_category_idx" ON "vault_entries" USING btree ("user_id","category");
